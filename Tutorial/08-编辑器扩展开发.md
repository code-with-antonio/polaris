## 8.2 StateField - 编辑器状态

StateField 用于在 EditorState 中存储自定义状态。

### 基本概念

```typescript
import { StateField } from "@codemirror/state";

// 定义一个存储布尔值的 StateField
const myField = StateField.define<boolean>({
  // 创建初始状态
  create() {
    return false;
  },

  // 更新状态（每个事务都会调用）
  update(value, transaction) {
    // 处理 effects
    for (const effect of transaction.effects) {
      if (effect.is(myEffect)) {
        return effect.value;
      }
    }
    // 处理其他变化（如选区变化）
    if (transaction.selection) {
      return true;  // 选区变化时设为 true
    }
    return value;
  },

  // 提供给其他扩展使用
  provide: (field) => {
    return someExtension.from(field);
  },
});
```

### Polaris 实战：quickEditState

```typescript
// src/features/editor/extensions/quick-edit/index.ts
import { StateField, StateEffect } from "@codemirror/state";

// 定义 Effect
export const showQuickEditEffect = StateEffect.define<boolean>();

// 定义 StateField
export const quickEditState = StateField.define<boolean>({
  create() {
    return false;  // 初始状态：未激活
  },

  update(value, transaction) {
    // 处理 showQuickEditEffect
    for (const effect of transaction.effects) {
      if (effect.is(showQuickEditEffect)) {
        return effect.value;
      }
    }
    // 选区为空时关闭
    if (transaction.selection) {
      const selection = transaction.state.selection.main;
      if (selection.empty) {
        return false;
      }
    }
    return value;
  }
});
```

## 8.3 StateEffect - 状态更新消息

StateEffect 是一种安全的方式向 StateField 发送更新消息。

### 基本概念

```typescript
import { StateEffect } from "@codemirror/state";

// 定义一个 Effect 类型
const setValueEffect = StateEffect.define<string>();

// 使用 Effect 更新状态
view.dispatch({
  effects: setValueEffect.of("new value"),
});

// 在 StateField 中处理
const myField = StateField.define<string>({
  create() {
    return "initial";
  },
  update(value, transaction) {
    for (const effect of transaction.effects) {
      if (effect.is(setValueEffect)) {
        return effect.value;  // 返回 Effect 携带的值
      }
    }
    return value;
  }
});
```

### 多步骤状态管理

```typescript
// 定义多个 Effect
const setSuggestionEffect = StateEffect.define<string | null>();
const setLoadingEffect = StateEffect.define<boolean>();

// StateField 处理多个 Effect
const suggestionState = StateField.define<{
  text: string | null;
  isLoading: boolean;
}>({
  create() {
    return { text: null, isLoading: false };
  },
  update(state, transaction) {
    let newState = { ...state };
    for (const effect of transaction.effects) {
      if (effect.is(setSuggestionEffect)) {
        newState.text = effect.value;
        newState.isLoading = false;
      }
      if (effect.is(setLoadingEffect)) {
        newState.isLoading = effect.value;
      }
    }
    return newState;
  }
});
```

## 8.4 ViewPlugin - 视图逻辑

ViewPlugin 用于处理视图相关逻辑，如事件监听、定时器等。

### 基本概念

```typescript
import { ViewPlugin } from "@codemirror/view";

// 定义 Plugin
const myPlugin = ViewPlugin.fromClass(
  class {
    // 构造函数：初始化
    constructor(view: EditorView) {
      // 设置事件监听
    }

    // 更新回调：每次更新都会调用
    update(update: ViewUpdate) {
      if (update.docChanged) {
        // 文档变化处理
      }
      if (update.selectionSet) {
        // 选区变化处理
      }
    }

    // 销毁：清理资源
    destroy() {
      // 清理定时器、事件监听等
    }
  }
);
```

### Polaris 实战：Debounce Plugin

```typescript
// src/features/editor/extensions/suggestion/index.ts
const createDebouncePlugin = (fileName: string) => {
  return ViewPlugin.fromClass(
    class {
      constructor(view: EditorView) {
        this.triggerSuggestion(view);
      }

      update(update: ViewUpdate) {
        if (update.docChanged || update.selectionSet) {
          this.triggerSuggestion(update.view);
        }
      }

      triggerSuggestion(view: EditorView) {
        // 清除之前的定时器
        if (debounceTimer !== null) {
          clearTimeout(debounceTimer);
        }

        // 取消之前的请求
        if (currentAbortController !== null) {
          currentAbortController.abort();
        }

        // 设置新的定时器
        debounceTimer = window.setTimeout(async () => {
          const payload = generatePayload(view, fileName);
          if (!payload) return;

          currentAbortController = new AbortController();
          const suggestion = await fetcher(payload, currentAbortController.signal);

          // 更新状态
          view.dispatch({
            effects: setSuggestionEffect.of(suggestion),
          });
        }, DEBOUNCE_DELAY);
      }

      destroy() {
        if (debounceTimer !== null) {
          clearTimeout(debounceTimer);
        }
        if (currentAbortController !== null) {
          currentAbortController.abort();
        }
      }
    }
  );
};
```

## 8.5 Decoration - 自定义元素

Decoration 用于在编辑器中渲染自定义元素。

### Decoration 类型

| 类型 | 说明 |
|------|------|
| `Decoration.mark()` | 内联标记（如高亮） |
| `Decoration.widget()` | 行内小部件（如图标） |
| `Decoration.line()` | 整行装饰 |
| `Decoration.set()` | 装饰集合 |

### Widget 示例

```typescript
import { Decoration, WidgetType } from "@codemirror/view";

// 创建自定义 Widget
class SuggestionWidget extends WidgetType {
  constructor(readonly text: string) {
    super();
  }

  toDOM() {
    const span = document.createElement("span");
    span.textContent = this.text;
    span.style.opacity = "0.4";  // 幽灵文本效果
    span.style.pointerEvents = "none";
    return span;
  }
}

// 创建装饰
const decoration = Decoration.widget({
  widget: new SuggestionWidget("suggested text"),
  side: 1,  // 1 = 在光标后，-1 = 在光标前
}).range(cursorPosition);
```

### Polaris 实战：Suggestion 装饰

```typescript
// src/features/editor/extensions/suggestion/index.ts
const renderPlugin = ViewPlugin.fromClass(
  class {
    decorations: DecorationSet;

    constructor(view: EditorView) {
      this.decorations = this.build(view);
    }

    update(update: ViewUpdate) {
      // 检查是否需要重建
      const suggestionChanged = update.transactions.some((transaction) => {
        return transaction.effects.some((effect) => {
          return effect.is(setSuggestionEffect);
        });
      });

      const shouldRebuild =
        update.docChanged || update.selectionSet || suggestionChanged;

      if (shouldRebuild) {
        this.decorations = this.build(update.view);
      }
    }

    build(view: EditorView) {
      const suggestion = view.state.field(suggestionState);
      if (!suggestion) {
        return Decoration.none;
      }

      // 在光标位置创建 Widget
      const cursor = view.state.selection.main.head;
      return Decoration.set([
        Decoration.widget({
          widget: new SuggestionWidget(suggestion),
          side: 1,
        }).range(cursor),
      ]);
    }
  },
  { decorations: (plugin) => plugin.decorations }
);
```

## 8.6 Tooltip - 悬浮提示

Tooltip 用于在特定位置显示悬浮内容。

### Tooltip 结构

```typescript
interface Tooltip {
  pos: number;           // 位置
  above?: boolean;        // 显示在上方还是下方
  strictSide?: boolean;   // 是否严格在某一侧
  create(): { dom: Element };  // 创建 DOM
}
```

### Polaris 实战：Selection Tooltip

```typescript
// src/features/editor/extensions/selection-tooltip.ts
const createTooltipForSelection = (state: EditorState): readonly Tooltip[] => {
  const selection = state.selection.main;

  // 无选区时不显示
  if (selection.empty) {
    return [];
  }

  return [
    {
      pos: selection.to,       // 显示在选区末尾
      above: false,            // 显示在下方
      strictSide: false,       // 可以根据空间调整
      create() {
        const dom = document.createElement("div");
        dom.className = "bg-popover p-2 rounded border shadow-md flex gap-2";

        // 添加按钮
        const addToChatButton = document.createElement("button");
        addToChatButton.textContent = "Add to Chat";
        // ...

        dom.appendChild(addToChatButton);
        return { dom };
      },
    }
  ];
};

// 定义 StateField 提供 Tooltip
const selectionTooltipField = StateField.define<readonly Tooltip[]>({
  create(state) {
    return createTooltipForSelection(state);
  },

  update(tooltips, transaction) {
    if (transaction.docChanged || transaction.selection) {
      return createTooltipForSelection(transaction.state);
    }
    return tooltips;
  },

  // 通过 showTooltip 提供给 EditorView
  provide: (field) => showTooltip.computeN(
    [field],
    (state) => state.field(field),
  ),
});

export const selectionTooltip = () => [
  selectionTooltipField,
  captureViewExtension,  // 捕获 EditorView 引用
];
```

## 8.7 Keymap - 快捷键

### 自定义快捷键

```typescript
import { keymap } from "@codemirror/view";

const customKeymap = keymap.of([
  {
    key: "Mod-k",  // Ctrl/Cmd + K
    run: (view) => {
      const selection = view.state.selection.main;
      if (selection.empty) {
        return false;  // 无选区，不处理
      }

      // 触发 Quick Edit
      view.dispatch({
        effects: showQuickEditEffect.of(true),
      });
      return true;  // 已处理
    },
  },
  {
    key: "Tab",
    run: (view) => {
      // 检查是否有建议
      const suggestion = view.state.field(suggestionState);
      if (!suggestion) {
        return false;  // 让 Tab 执行默认行为（缩进）
      }

      // 接受建议
      const cursor = view.state.selection.main.head;
      view.dispatch({
        changes: { from: cursor, insert: suggestion },
        selection: { anchor: cursor + suggestion.length },
        effects: setSuggestionEffect.of(null),
      });
      return true;
    },
  },
]);
```

### Keymap 优先级

```typescript
// 多个 keymap 会按顺序执行
keymap.of([...myKeymaps]),  // 自定义优先
keymap.of([...defaultKeymaps]),  // 默认快捷键
```

## 8.8 完整扩展示例

### Quick Edit 扩展结构

```typescript
// src/features/editor/extensions/quick-edit/index.ts

// 1. StateEffect：触发状态变化
export const showQuickEditEffect = StateEffect.define<boolean>();

// 2. StateField：存储激活状态
export const quickEditState = StateField.define<boolean>({
  create() { return false; },
  update(value, transaction) {
    for (const effect of transaction.effects) {
      if (effect.is(showQuickEditEffect)) {
        return effect.value;
      }
    }
    if (transaction.selection?.main.empty) {
      return false;
    }
    return value;
  }
});

// 3. Tooltip 创建函数
const createQuickEditTooltip = (state: EditorState): readonly Tooltip[] => {
  // ... 创建编辑表单的 Tooltip
};

// 4. StateField 提供 Tooltip
const quickEditTooltipField = StateField.define<readonly Tooltip[]>({
  create(state) { return createQuickEditTooltip(state); },
  update(tooltips, transaction) {
    // ... 更新逻辑
  },
  provide: (field) => showTooltip.computeN([field], (state) => state.field(field)),
});

// 5. Keymap
const quickEditKeymap = keymap.of([
  {
    key: "Mod-k",
    run: (view) => {
      const selection = view.state.selection.main;
      if (selection.empty) return false;
      view.dispatch({ effects: showQuickEditEffect.of(true) });
      return true;
    },
  },
]);

// 6. 捕获 EditorView 引用
const captureViewExtension = EditorView.updateListener.of((update) => {
  editorView = update.view;
});

// 7. 导出组合扩展
export const quickEdit = (fileName: string) => [
  quickEditState,
  quickEditTooltipField,
  quickEditKeymap,
  captureViewExtension,
];
```

## 8.9 ViewUpdate - 更新信息

ViewUpdate 包含每次更新的详细信息。

### 主要属性

```typescript
interface ViewUpdate {
  // 变化相关
  docChanged: boolean;                    // 文档是否变化
  selectionSet: boolean;                  // 选区是否变化
  viewportChanged: boolean;               // 视口是否变化
  heightChanged: boolean;                 // 高度是否变化

  // 事务
  transactions: Transaction[];            // 本次更新的所有事务

  // 状态
  state: EditorState;                    // 新的 EditorState
  startState: EditorState;               // 更新前的 EditorState

  // 视图
  view: EditorView;                      // EditorView 实例
}
```

### 使用示例

```typescript
const myPlugin = ViewPlugin.fromClass(
  class {
    update(update: ViewUpdate) {
      // 文档变化
      if (update.docChanged) {
        const newContent = update.state.doc.toString();
        console.log("Document changed:", newContent);
      }

      // 选区变化
      if (update.selectionSet) {
        const sel = update.state.selection.main;
        console.log("Selection:", sel.from, "-", sel.to);
      }

      // 遍历事务中的 effects
      for (const tr of update.transactions) {
        for (const effect of tr.effects) {
          if (effect.is(myEffect)) {
            console.log("My effect:", effect.value);
          }
        }
      }
    }
  }
);
```

## 8.10 EditorView 实例捕获

某些扩展需要直接操作 EditorView，但 StateField/ViewPlugin 的回调中不能直接访问 view。

### 问题

```typescript
// ❌ 错误：在 StateField 的 create/update 中无法访问 EditorView
const badField = StateField.define({
  create() {
    view.dispatch({ ... });  // 错误：view 不存在
    return null;
  }
});
```

### 解决方案：捕获引用

```typescript
// src/features/editor/extensions/selection-tooltip.ts
let editorView: EditorView | null = null;

// 通过 EditorView.updateListener 捕获引用
const captureViewExtension = EditorView.updateListener.of((update) => {
  editorView = update.view;
});

// 在 Tooltip 的 onclick 中使用
quickEditButton.onclick = () => {
  if (editorView) {
    editorView.dispatch({
      effects: showQuickEditEffect.of(true),
    });
  }
};

// 导出时包含捕获扩展
export const selectionTooltip = () => [
  selectionTooltipField,
  captureViewExtension,
];
```

```

│                    suggestion 扩展                    │

│  └─ 提供给 EditorView                              │
│                                                     │
│  Keymap                                             │
│  └─ Tab 键接受建议                                  │
│                                                     │
└─────────────────────────────────────────────────────┘
```

### 生成请求 Payload

```typescript
const generatePayload = (view: EditorView, fileName: string) => {
  const code = view.state.doc.toString();
  const cursorPosition = view.state.selection.main.head;
  const currentLine = view.state.doc.lineAt(cursorPosition);

  return {
    fileName,
    code,
    currentLine: currentLine.text,
    textBeforeCursor: currentLine.text.slice(0, cursorPosition - currentLine.from),
    textAfterCursor: currentLine.text.slice(cursorPosition - currentLine.from),
    lineNumber: currentLine.number,
    previousLines: /* 前5行 */,
    nextLines: /* 后5行 */,
  };
};
```
